cmake_minimum_required(VERSION 3.8)
project(engineer_mtc)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_task_constructor_msgs REQUIRED)
find_package(moveit_task_constructor_core REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(Eigen3 REQUIRED)

add_library(grasp_geometry src/grasp_geometry.cpp)
target_include_directories(grasp_geometry PUBLIC include)
target_link_libraries(grasp_geometry PUBLIC Eigen3::Eigen)
ament_target_dependencies(grasp_geometry PUBLIC geometry_msgs)
add_executable(engineer_mtc_node src/main.cpp src/mtc_node.cpp src/pregrasp_task.cpp)
target_include_directories(engineer_mtc_node PRIVATE include)
ament_target_dependencies(engineer_mtc_node rclcpp geometry_msgs sensor_msgs std_srvs moveit_msgs
  moveit_core moveit_task_constructor_msgs moveit_task_constructor_core tf2_ros tf2_eigen)
target_link_libraries(engineer_mtc_node grasp_geometry)
install(TARGETS engineer_mtc_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/demo_pregrasp.py DESTINATION lib/${PROJECT_NAME} RENAME demo_pregrasp)
ament_package()
