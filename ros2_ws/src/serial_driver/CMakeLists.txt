cmake_minimum_required(VERSION 3.8)
project(serial_driver)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(std_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Threads REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/HardwareStatus.msg" DEPENDENCIES std_msgs)
rosidl_get_typesupport_target(status_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)

add_library(serial_protocol SHARED src/protocol.cpp src/serial_port.cpp)
target_include_directories(serial_protocol PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
add_library(engineer_hardware SHARED src/engineer_hardware.cpp)
target_include_directories(engineer_hardware PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(engineer_hardware hardware_interface pluginlib rclcpp rclcpp_lifecycle std_msgs)
target_link_libraries(engineer_hardware serial_protocol "${status_typesupport}" Threads::Threads)
pluginlib_export_plugin_description_file(hardware_interface serial_driver.xml)

add_executable(hardware_supervisor src/hardware_supervisor.cpp)
ament_target_dependencies(hardware_supervisor rclcpp controller_manager_msgs)
target_link_libraries(hardware_supervisor "${status_typesupport}")
install(TARGETS serial_protocol engineer_hardware EXPORT export_serial_driver
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(TARGETS hardware_supervisor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/docs")
  install(DIRECTORY docs DESTINATION share/${PROJECT_NAME})
endif()
install(FILES 通信协议.md DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_serial_driver HAS_LIBRARY_TARGET)
ament_export_dependencies(hardware_interface pluginlib rclcpp rclcpp_lifecycle std_msgs rosidl_default_runtime)
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test/test_serial_driver.cpp")
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_serial_driver test/test_serial_driver.cpp TIMEOUT 30)
  target_link_libraries(test_serial_driver serial_protocol engineer_hardware util Threads::Threads)
  ament_target_dependencies(test_serial_driver rclcpp hardware_interface)
endif()
ament_package()
