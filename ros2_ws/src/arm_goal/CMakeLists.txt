cmake_minimum_required(VERSION 3.8)
project(arm_goal)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(action_msgs REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(act_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(jsoncpp REQUIRED)
include_directories(include)
add_library(goal_options src/goal_options.cpp)
ament_target_dependencies(goal_options rclcpp rclcpp_action act_interfaces std_msgs)
add_executable(plan_goal src/plan_goal_main.cpp src/goal_client_node.cpp src/topic_input.cpp src/config_arguments.cpp)
ament_target_dependencies(plan_goal ament_index_cpp action_msgs rclcpp rclcpp_action act_interfaces std_msgs moveit_msgs)
target_link_libraries(plan_goal goal_options JsonCpp::JsonCpp)
install(TARGETS plan_goal DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_goal_options test/test_goal_options.cpp)
  target_link_libraries(test_goal_options goal_options)
  ament_target_dependencies(test_goal_options rclcpp rclcpp_action act_interfaces std_msgs)
endif()
ament_package()
