cmake_minimum_required(VERSION 3.8)
project(arm_planning)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "action/PlanMotion.action"
  DEPENDENCIES geometry_msgs moveit_msgs)
add_executable(arm_planning_node src/arm_planning_node.cpp)
ament_target_dependencies(arm_planning_node rclcpp rclcpp_action geometry_msgs moveit_msgs
  moveit_core moveit_ros_planning_interface)
rosidl_get_typesupport_target(action_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(arm_planning_node "${action_typesupport}")
install(TARGETS arm_planning_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/plan_goal.py DESTINATION lib/${PROJECT_NAME} RENAME plan_goal)
ament_export_dependencies(rosidl_default_runtime)
ament_package()
