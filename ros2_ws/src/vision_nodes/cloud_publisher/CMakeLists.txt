cmake_minimum_required(VERSION 3.8)
project(model_cloud_publisher)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(vision_core REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)

add_executable(model_cloud_publisher_node src/main.cpp src/model_cloud_publisher_node.cpp)
target_compile_features(model_cloud_publisher_node PUBLIC cxx_std_17)
target_include_directories(model_cloud_publisher_node PRIVATE include)
ament_target_dependencies(model_cloud_publisher_node rclcpp ament_index_cpp sensor_msgs pcl_conversions)
target_link_libraries(model_cloud_publisher_node vision_core::vision_core)
install(TARGETS model_cloud_publisher_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/../../../../assets/energy/energy.ply
  DESTINATION share/${PROJECT_NAME}/assets/energy)
ament_package()
