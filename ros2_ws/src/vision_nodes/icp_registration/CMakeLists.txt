cmake_minimum_required(VERSION 3.8)
project(icp_registration)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(vision_core REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(vision_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(pcl_conversions REQUIRED)

add_executable(icp_registration_node src/main.cpp src/icp_registration_node.cpp)
target_compile_features(icp_registration_node PUBLIC cxx_std_17)
target_include_directories(icp_registration_node PRIVATE include)
ament_target_dependencies(icp_registration_node rclcpp sensor_msgs std_msgs geometry_msgs vision_msgs
  tf2_ros tf2_eigen pcl_conversions)
target_link_libraries(icp_registration_node vision_core::vision_core)
install(TARGETS icp_registration_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_projection_roi test/test_projection_roi.cpp src/icp_registration_node.cpp)
  target_compile_features(test_projection_roi PUBLIC cxx_std_17)
  target_include_directories(test_projection_roi PRIVATE include)
  ament_target_dependencies(test_projection_roi rclcpp sensor_msgs std_msgs geometry_msgs vision_msgs
    tf2_ros tf2_eigen pcl_conversions)
  target_link_libraries(test_projection_roi vision_core::vision_core)
endif()
ament_package()
