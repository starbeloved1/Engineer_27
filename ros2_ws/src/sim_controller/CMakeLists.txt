cmake_minimum_required(VERSION 3.8)
project(sim_controller)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(engineer_mtc REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(std_srvs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(mtc_panel SHARED src/mtc_panel.cpp include/sim_controller/mtc_panel.hpp)
target_include_directories(mtc_panel PRIVATE include)
target_link_libraries(mtc_panel Qt5::Widgets)
ament_target_dependencies(mtc_panel engineer_mtc pluginlib rclcpp rviz_common std_srvs)
pluginlib_export_plugin_description_file(rviz_common rviz_plugins.xml)
install(TARGETS mtc_panel LIBRARY DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
install(PROGRAMS energy_display_node DESTINATION lib/${PROJECT_NAME})
ament_package()
